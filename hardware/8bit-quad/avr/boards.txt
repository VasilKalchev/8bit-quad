##   Menu options are in "PascalCase".                                       ##
##                                                                           ##
## Notes:                                                                    ##
## Menu items with "[F]" appended to them, indicate that the program must be ##
## uploaded with a programmer, in order for the setting to take effect.      ##

## Menus items
menu.CORE=Core
menu.CLK_FREQ=Clock frequency
menu.UP_SPEED=Upload speed [F]
menu.BOD=Brown out detector [F]
menu.BL=Bootloader [F]
menu.EEPROM=EEPROM [F]
menu.OPTIMIZATION=Optimization
menu.SOLUTION=Solution
## -----------


## Flight controller ##########################################################
fc.name=8bit-quad_fc

fc.build.board=BOARD_8BIT_QUAD_FC
fc.build.mcu=atmega328p
fc.build.variant=standard

fc.upload.tool=avrdude
fc.upload.maximum_data_size=2048

fc.bootloader.tool=avrdude
fc.bootloader.unlock_bits=0x3F
fc.bootloader.lock_bits=0x0F
fc.bootloader.low_fuses=0xFF

fc.serial.disableDTR=false
fc.serial.disableRTS=false


## CORE -----------------------------------------------------------------------
fc.build.core=swap_tmr0_with_tmr2

fc.menu.CORE.SwapTmr2WithTmr0=Swap TMR2 with TMR0 (default)
fc.menu.CORE.SwapTmr2WithTmr0.build.core=swap_tmr0_with_tmr2
## ----------------------------------------------------------------------------

## CLK_FREQ -------------------------------------------------------------------
fc.menu.CLK_FREQ.16MHz=16MHz

fc.menu.CLK_FREQ.20MHz=20MHz
fc.menu.CLK_FREQ.20MHz.build.f_cpu=20000000L
## ----------------------------------------------------------------------------

## UP_SPEED -------------------------------------------------------------------

fc.menu.UP_SPEED.250000.upload.speed=250000
## ----------------------------------------------------------------------------

## BOD ------------------------------------------------------------------------

fc.menu.BOD.1V8=1.8V
fc.menu.BOD.1V8.bootloader.extended_fuses=0xFE

fc.menu.BOD.Disabled=Disabled
fc.menu.BOD.Disabled.bootloader.extended_fuses=0xFF
## ---------------------------------------------------------------------------

## EEPROM ---------------------------------------------------------------------
fc.menu.EEPROM.keep=Preserve
fc.menu.EEPROM.keep.bootloader.eesave_bit=0

fc.menu.EEPROM.erase=Erase (only when using programmer)
fc.menu.EEPROM.erase.bootloader.eesave_bit=1
## ----------------------------------------------------------------------------

## BL -------------------------------------------------------------------------
fc.menu.BL.Optiboot=Optiboot
fc.menu.BL.Optiboot.bootloader.file=optiboot/optiboot_{build.mcu}_{build.f_cpu}_{upload.speed}baud.hex
fc.menu.BL.Optiboot.bootloader.high_fuses=0b1101{bootloader.eesave_bit}110
fc.menu.BL.Optiboot.upload.maximum_size=32256
fc.menu.BL.Optiboot.upload.protocol=arduino

fc.menu.BL.None=None
fc.menu.BL.None.bootloader.high_fuses=0b1101{bootloader.eesave_bit}111
fc.menu.BL.None.upload.maximum_size=32768
## ----------------------------------------------------------------------------

## OPTIMIZATION----------------------------------------------------------------

fc.menu.OPTIMIZATION.O0.build.optimization=-O0

fc.menu.OPTIMIZATION.O3.build.optimization=-O3
